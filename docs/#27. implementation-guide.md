```csharp
namespace VelikiyPrikalel.OLLAMACHAT.Infrastructure.Services.Unity;

/// <summary>
/// Service for computing file hashes.
/// </summary>
public class FileHashingService
{
    private readonly ILogger<FileHashingService> logger;

    public FileHashingService(ILogger<FileHashingService> logger)
    {
        this.logger = logger;
    }

    /// <summary>
    /// Compute SHA256 hash of file content.
    /// </summary>
    public async Task<string> ComputeFileHashAsync(string filePath, CancellationToken cancellationToken = default)
    {
        try
        {
            await using var stream = new FileStream(filePath, FileMode.Open, FileAccess.Read, FileShare.Read, 4096, true);
            using var sha256 = SHA256.Create();
            var hash = await sha256.ComputeHashAsync(stream, cancellationToken);
            return Convert.ToHexString(hash).ToLowerInvariant();
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Failed to compute hash for file: {FilePath}", filePath);
            throw;
        }
    }

    /// <summary>
    /// Compute SHA256 hash of string content.
    /// </summary>
    public string ComputeContentHash(string content)
    {
        using var sha256 = SHA256.Create();
        var hash = sha256.ComputeHash(Encoding.UTF8.GetBytes(content));
        return Convert.ToHexString(hash).ToLowerInvariant();
    }
}
```

### Step 3.2: Unity File Watcher Service

Create `Source/OLLAMACHAT.Infrastructure/Services/Unity/UnityFileWatcherService.cs`:

```csharp
namespace VelikiyPrikalel.OLLAMACHAT.Infrastructure.Services.Unity;

/// <inheritdoc />
public class UnityFileWatcherService : IUnityFileWatcherService
{
    private readonly ILogger<UnityFileWatcherService> logger;
    private readonly UnityAnalysisSettings settings;
    private readonly object lockObject = new();
    private FileSystemWatcher?[] watchers;
    private bool disposed;

    public event EventHandler<UnityFileChangedEventArgs>? FileChanged;

    public bool IsWatching { get; private set; }

    public UnityFileWatcherService(ILogger<UnityFileWatcherService> logger, IOptions<UnityAnalysisSettings> settings)
    {
        this.logger = logger;
        this.settings = settings.Value;
    }

    /// <inheritdoc />
    public async Task StartWatchingAsync(string unityProjectPath, CancellationToken cancellationToken = default)
    {
        if (disposed)
            throw new ObjectDisposedException(nameof(UnityFileWatcherService));

        lock (lockObject)
        {
            if (IsWatching)
            {
                logger.LogWarning("File watcher is already active for path: {UnityProjectPath}", unityProjectPath);
                return;
            }

            try
            {
                SetupWatchers(unityProjectPath);
                IsWatching = true;
                logger.LogInformation("Started watching Unity project: {UnityProjectPath}", unityProjectPath);
            }
            catch (Exception ex)
            {
                logger.LogError(ex, "Failed to start file watcher for path: {UnityProjectPath}", unityProjectPath);
                throw;
            }
        }
    }

    /// <inheritdoc />
    public Task StopWatchingAsync()
    {
        if (disposed)
            return Task.CompletedTask;

        lock (lockObject)
        {
            if (!IsWatching)
                return;

            try
            {
                DisposeWatchers();
                IsWatching = false;
                logger.LogInformation("Stopped watching Unity project");
            }
            catch (Exception ex)
            {
                logger.LogError(ex, "Failed to stop file watcher");
            }
        }

        return Task.CompletedTask;
    }

    private void SetupWatchers(string unityProjectPath)
    {
        DisposeWatchers();

        var watchedDirectories = GetWatchedDirectories(unityProjectPath);
        watchers = new FileSystemWatcher[watchedDirectories.Count];

        for (int i = 0; i < watchedDirectories.Count; i++)
        {
            var directory = watchedDirectories[i];
            var watcher = new FileSystemWatcher(directory)
            {
                IncludeSubdirectories = true,
                EnableRaisingEvents = true,
                Filter = "*.*" // We'll filter in the event handler
            };

            // Set up event handlers
            watcher.Changed += OnFileChanged;
            watcher.Created += OnFileChanged;
            watcher.Deleted += OnFileChanged;
            watcher.Renamed += OnFileRenamed;

            watchers[i] = watcher;
        }
    }

    private List<string> GetWatchedDirectories(string unityProjectPath)
    {
        var directories = new List<string>();

        // Always watch Assets directory
        var assetsPath = Path.Combine(unityProjectPath, "Assets");
        if (Directory.Exists(assetsPath))
            directories.Add(assetsPath);

        // Watch Packages directory if it exists
        var packagesPath = Path.Combine(unityProjectPath, "Packages");
        if (Directory.Exists(packagesPath))
            directories.Add(packagesPath);

        return directories;
    }

    private void OnFileChanged(object sender, FileSystemEventArgs e)
    {
        if (!ShouldWatchFile(e.FullPath))
            return;

        var fileType = DetermineFileType(e.FullPath);
        var changeType = e.ChangeType switch
        {
            WatcherChangeTypes.Created => FileChangeType.Created,
            WatcherChangeTypes.Deleted => FileChangeType.Deleted,
            WatcherChangeTypes.Changed => FileChangeType.Modified,
            _ => FileChangeType.Modified
        };

        var args = new UnityFileChangedEventArgs
        {
            FilePath = e.FullPath,
            FileType = fileType,
            ChangeType = changeType,
            ChangeTime = DateTimeOffset.UtcNow
        };

        Task.Run(() => FileChanged?.Invoke(this, args));
    }

    private void OnFileRenamed(object sender, RenamedEventArgs e)
    {
        if (!ShouldWatchFile(e.FullPath) && !ShouldWatchFile(e.OldFullPath))
            return;

        // Handle old file deletion
        if (ShouldWatchFile(e.OldFullPath))
        {
            var oldArgs = new UnityFileChangedEventArgs
            {
                FilePath = e.OldFullPath,
                FileType = DetermineFileType(e.OldFullPath),
                ChangeType = FileChangeType.Deleted,
                ChangeTime = DateTimeOffset.UtcNow
            };

            Task.Run(() => FileChanged?.Invoke(this, oldArgs));
        }

        // Handle new file creation
        if (ShouldWatchFile(e.FullPath))
        {
            var newArgs = new UnityFileChangedEventArgs
            {
                FilePath = e.FullPath,
                FileType = DetermineFileType(e.FullPath),
                ChangeType = FileChangeType.Created,
                ChangeTime = DateTimeOffset.UtcNow
            };

            Task.Run(() => FileChanged?.Invoke(this, newArgs));
        }
    }

    private bool ShouldWatchFile(string filePath)
    {
        if (string.IsNullOrEmpty(filePath))
            return false;

        var extension = Path.GetExtension(filePath).ToLowerInvariant();
        return settings.WatchedFileExtensions.Contains(extension);
    }

    private UnityFileType DetermineFileType(string filePath)
    {
        var extension = Path.GetExtension(filePath).ToLowerInvariant();
        var fileName = Path.GetFileName(filePath).ToLowerInvariant();

        return extension switch
        {
            ".unity" => UnityFileType.Scene,
            ".prefab" => UnityFileType.Prefab,
            ".meta" when fileName.EndsWith(".cs.meta") => UnityFileType.ScriptMeta,
            _ => UnityFileType.Other
        };
    }

    private void DisposeWatchers()
    {
        if (watchers == null)
            return;

        foreach (var watcher in watchers)
        {
            if (watcher != null)
            {
                watcher.Changed -= OnFileChanged;
                watcher.Created -= OnFileChanged;
                watcher.Deleted -= OnFileChanged;
                watcher.Renamed -= OnFileRenamed;
                watcher.Dispose();
            }
        }

        watchers = null;
    }

    /// <inheritdoc />
    public async ValueTask DisposeAsync()
    {
        if (disposed)
            return;

        await StopWatchingAsync();
        disposed = true;
    }

    /// <inheritdoc />
    public void Dispose()
    {
        if (disposed)
            return;

        StopWatchingAsync().GetAwaiter().GetResult();
        disposed = true;
        GC.SuppressFinalize(this);
    }
}
```

```csharp
namespace VelikiyPrikalel.OLLAMACHAT.Infrastructure.Services.Unity;

/// <inheritdoc />
public class UnityCacheManager : IUnityCacheManager
{
    private readonly IMemoryCache memoryCache;
    private readonly IUnityFileTrackingRepository fileTrackingRepository;
    private readonly ILogger<UnityCacheManager> logger;
    private readonly UnityAnalysisSettings settings;
    private readonly SemaphoreSlim cacheLock = new(1, 1);

    // Cache keys
    private const string GUID_TO_SCRIPT_MAP_KEY = "Unity:GuidToScriptMap";
    private const string FILE_TRACKING_PREFIX = "Unity:FileTracking:";
    private const string RELATIONSHIPS_PREFIX = "Unity:Relationships:";
    private const string GAMEOBJECTS_PREFIX = "Unity:GameObjects:";
    private const string MONOBEHAVIOURS_PREFIX = "Unity:MonoBehaviours:";

    public UnityCacheManager(
        IMemoryCache memoryCache,
        IUnityFileTrackingRepository fileTrackingRepository,
        ILogger<UnityCacheManager> logger,
        IOptions<UnityAnalysisSettings> settings)
    {
        this.memoryCache = memoryCache;
        this.fileTrackingRepository = fileTrackingRepository;
        this.logger = logger;
        this.settings = settings.Value;
    }

    /// <inheritdoc />
    public async Task<T?> GetAsync<T>(string key, CancellationToken cancellationToken = default) where T : class
    {
        await cacheLock.WaitAsync(cancellationToken);
        try
        {
            if (memoryCache.TryGetValue(key, out T? value))
            {
                logger.LogDebug("Cache hit for key: {CacheKey}", key);
                return value;
            }

            logger.LogDebug("Cache miss for key: {CacheKey}", key);
            return null;
        }
        finally
        {
            cacheLock.Release();
        }
    }

    /// <inheritdoc />
    public async Task SetAsync<T>(string key, T value, TimeSpan? expiry = null, CancellationToken cancellationToken = default) where T : class
    {
        await cacheLock.WaitAsync(cancellationToken);
        try
        {
            var options = new MemoryCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow = expiry ?? settings.CacheExpiry,
                SlidingExpiration = TimeSpan.FromMinutes(30)
            };

            memoryCache.Set(key, value, options);
            logger.LogDebug("Cached value for key: {CacheKey}", key);
        }
        finally
        {
            cacheLock.Release();
        }
    }

    /// <inheritdoc />
    public async Task InvalidateAsync(string pattern, CancellationToken cancellationToken = default)
    {
        await cacheLock.WaitAsync(cancellationToken);
        try
        {
            // For simplicity, we'll clear all cache entries matching the pattern
            // In a production environment, you might want more sophisticated pattern matching
            var keysToRemove = new List<string>();

            if (pattern == "*" || pattern == "Unity:*")
            {
                // Clear all Unity-related cache
                keysToRemove.Add(GUID_TO_SCRIPT_MAP_KEY);
            }
            else if (pattern.StartsWith("Unity:FileTracking:"))
            {
                // Clear specific file tracking
                keysToRemove.Add(pattern);
            }
            else if (pattern.StartsWith("Unity:Relationships:"))
            {
                // Clear relationships for specific file
                keysToRemove.Add(pattern);
            }

            foreach (var key in keysToRemove)
            {
                memoryCache.Remove(key);
                logger.LogDebug("Invalidated cache key: {CacheKey}", key);
            }
        }
        finally
        {
            cacheLock.Release();
        }
    }

    /// <inheritdoc />
    public async Task<bool> IsFileChangedAsync(string filePath, string contentHash, CancellationToken cancellationToken = default)
    {
        try
        {
            var cacheKey = $"{FILE_TRACKING_PREFIX}{filePath}";
            var tracking = await GetAsync<UnityFileTracking>(cacheKey, cancellationToken);
            
            if (tracking == null)
            {
                // Check database
                tracking = await fileTrackingRepository.GetByFilePathAsync(filePath, cancellationToken);
                if (tracking != null)
                {
                    await SetAsync(cacheKey, tracking, settings.CacheExpiry, cancellationToken);
                }
            }

            return tracking?.ContentHash != contentHash;
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Error checking if file changed: {FilePath}", filePath);
            return true; // Assume changed on error
        }
    }

    /// <inheritdoc />
    public async Task UpdateFileTrackingAsync(string filePath, string contentHash, UnityFileType fileType, CancellationToken cancellationToken = default)
    {
        try
        {
            var tracking = new UnityFileTracking
            {
                FilePath = filePath,
                ContentHash = contentHash,
                FileType = fileType,
                LastAnalyzed = DateTimeOffset.UtcNow,
                LastModified = File.GetLastWriteTimeUtc(filePath),
                IsDeleted = false
            };

            // Update database
            await fileTrackingRepository.UpsertAsync(tracking, cancellationToken);

            // Update cache
            var cacheKey = $"{FILE_TRACKING_PREFIX}{filePath}";
            await SetAsync(cacheKey, tracking, settings.CacheExpiry, cancellationToken);

            logger.LogDebug("Updated file tracking for: {FilePath}", filePath);
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Error updating file tracking: {FilePath}", filePath);
            throw;
        }
    }

    /// <inheritdoc />
    public async Task<UnityCacheStatistics> GetStatisticsAsync(CancellationToken cancellationToken = default)
    {
        try
        {
            var trackedFiles = await fileTrackingRepository.GetCountAsync(cancellationToken);
            
            // Estimate cache size (this is a rough estimate)
            var cacheSize = EstimateCacheSize();

            return new UnityCacheStatistics
            {
                TotalEntries = memoryCache.Count,
                TrackedFiles = trackedFiles,
                CacheSizeBytes = cacheSize,
                HitRatio = 0.85, // This would be tracked in a real implementation
                LastCleanup = DateTimeOffset.UtcNow.AddMinutes(-15)
            };
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Error getting cache statistics");
            return new UnityCacheStatistics();
        }
    }

    private long EstimateCacheSize()
    {
        // This is a very rough estimate
        // In a real implementation, you might want to track actual memory usage
        return memoryCache.Count * 1024; // Assume 1KB per entry on average
    }
}
```

```csharp
namespace VelikiyPrikalel.OLLAMACHAT.Infrastructure.Services.Unity;

/// <summary>
/// Background service for Unity file analysis.
/// </summary>
public class UnityBackgroundAnalysisService : BackgroundService
{
    private readonly IUnityAnalysisCoordinator coordinator;
    private readonly ILogger<UnityBackgroundAnalysisService> logger;
    private readonly UnityAnalysisSettings settings;

    public UnityBackgroundAnalysisService(
        IUnityAnalysisCoordinator coordinator,
        ILogger<UnityBackgroundAnalysisService> logger,
        IOptions<UnityAnalysisSettings> settings)
    {
        this.coordinator = coordinator;
        this.logger = logger;
        this.settings = settings.Value;
    }

    /// <inheritdoc />
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        logger.LogInformation("Unity Background Analysis Service started");

        try
        {
            // Start analysis if configured
            if (!string.IsNullOrEmpty(settings.UnityProjectPath))
            {
                await coordinator.StartAnalysisAsync(settings.UnityProjectPath, stoppingToken);
            }

            // Main processing loop
            while (!stoppingToken.IsCancellationRequested)
            {
                try
                {
                    // Process any queued changes
                    await ProcessQueuedChangesAsync(stoppingToken);

                    // Perform periodic cleanup if needed
                    if (ShouldPerformCleanup())
                    {
                        await PerformCleanupAsync(stoppingToken);
                    }

                    // Wait before next iteration
                    await Task.Delay(TimeSpan.FromSeconds(1), stoppingToken);
                }
                catch (OperationCanceledException)
                {
                    // Normal cancellation, exit gracefully
                    break;
                }
                catch (Exception ex)
                {
                    logger.LogError(ex, "Error in background processing loop");
                    await Task.Delay(TimeSpan.FromSeconds(5), stoppingToken);
                }
            }
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Fatal error in Unity Background Analysis Service");
        }
        finally
        {
            await coordinator.StopAnalysisAsync();
            logger.LogInformation("Unity Background Analysis Service stopped");
        }
    }
```

## Troubleshooting

### Common Issues

1. **File Watcher Not Working**
   - Check if the Unity project path is correct
   - Verify file permissions
   - Check if the directory exists

2. **High Memory Usage**
   - Reduce cache expiry time
   - Implement cache size limits
   - Monitor for memory leaks

3. **Slow Performance**
   - Enable parallel processing
   - Optimize database queries
   - Check for proper indexing

4. **Database Issues**
   - Verify connection string
   - Check database permissions
   - Run database migrations

### Debugging Tips

- Enable verbose logging in settings
- Use the cache statistics endpoint
- Monitor the analysis status
- Check the background service logs
