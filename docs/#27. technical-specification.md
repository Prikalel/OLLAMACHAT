/// <summary>
/// Конфигурация сущности UnityGameObject.
/// </summary>
internal sealed class UnityGameObjectConfiguration : IEntityTypeConfiguration<UnityGameObject>
{
    public void Configure(EntityTypeBuilder<UnityGameObject> builder)
    {
        builder.HasKey(u => u.Id);
        
        builder.Property(u => u.Id)
            .ValueGeneratedOnAdd();

        builder.Property(u => u.Name)
            .IsRequired()
            .HasMaxLength(256);

        builder.Property(u => u.ScenePath)
            .IsRequired()
            .HasMaxLength(1000);

        builder.Property(u => u.Type)
            .IsRequired()
            .HasMaxLength(20);

        builder.Property(u => u.ContentHash)
            .IsRequired()
            .HasMaxLength(64);

        // Indices
        builder.HasIndex(u => u.FileId)
            .HasDatabaseName("IX_UnityGameObjects_FileId");

        builder.HasIndex(u => u.ScenePath)
            .HasDatabaseName("IX_UnityGameObjects_ScenePath");

        builder.HasIndex(u => u.Name)
            .HasDatabaseName("IX_UnityGameObjects_Name");

        builder.HasIndex(u => new { u.ScenePath, u.FileId })
            .IsUnique()
            .HasDatabaseName("IX_UnityGameObjects_ScenePath_FileId");

        builder.HasIndex(u => u.LastModified)
            .HasDatabaseName("IX_UnityGameObjects_LastModified");

        builder.HasIndex(u => u.IsDeleted)
            .HasDatabaseName("IX_UnityGameObjects_IsDeleted");

        builder.ToTable("UnityGameObjects");
    }
}

/// <summary>
/// Конфигурация сущности UnityMonoBehaviour.
/// </summary>
internal sealed class UnityMonoBehaviourConfiguration : IEntityTypeConfiguration<UnityMonoBehaviour>
{
    public void Configure(EntityTypeBuilder<UnityMonoBehaviour> builder)
    {
        builder.HasKey(u => u.Id);
        
        builder.Property(u => u.Id)
            .ValueGeneratedOnAdd();

        builder.Property(u => u.ScriptGuid)
            .IsRequired()
            .HasMaxLength(32);

        builder.Property(u => u.ComponentTypeName)
            .IsRequired()
            .HasMaxLength(500);

        builder.Property(u => u.ScenePath)
            .IsRequired()
            .HasMaxLength(1000);

        builder.Property(u => u.ContentHash)
            .IsRequired()
            .HasMaxLength(64);

        // Foreign keys
        builder.HasOne<UnityGameObject>()
            .WithMany(g => g.MonoBehaviours)
            .HasForeignKey(u => u.GameObjectFileId)
            .HasPrincipalKey(g => g.FileId)
            .OnDelete(DeleteBehavior.Cascade);

        builder.HasOne<UnityScriptMapping>()
            .WithMany(m => m.MonoBehaviours)
            .HasForeignKey(u => u.ScriptGuid)
            .HasPrincipalKey(m => m.Guid)
            .OnDelete(DeleteBehavior.Restrict);

        // Indices
        builder.HasIndex(u => u.ScriptGuid)
            .HasDatabaseName("IX_UnityMonoBehaviours_ScriptGuid");

        builder.HasIndex(u => u.GameObjectFileId)
            .HasDatabaseName("IX_UnityMonoBehaviours_GameObjectFileId");

        builder.HasIndex(u => u.ComponentTypeName)
            .HasDatabaseName("IX_UnityMonoBehaviours_ComponentTypeName");

        builder.HasIndex(u => new { u.ScenePath, u.GameObjectFileId })
            .HasDatabaseName("IX_UnityMonoBehaviours_ScenePath_GameObjectFileId");

        builder.HasIndex(u => u.LastModified)
            .HasDatabaseName("IX_UnityMonoBehaviours_LastModified");

        builder.HasIndex(u => u.IsDeleted)
            .HasDatabaseName("IX_UnityMonoBehaviours_IsDeleted");
